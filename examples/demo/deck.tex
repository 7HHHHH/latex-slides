\lsChapter{02}
\begin{frame}{Every content slide pairs text with a visual}\slideid{visual}
\lead{Text explains the visual; the visual carries the evidence.}
\at{56}{\lsContentTop}{470}{\begin{tikzpicture}
\begin{axis}[width=430bp,height=262bp,ybar,bar width=26bp,ymin=0,ymax=100,
  symbolic x coords={Text only,Text + figure},xtick=data,ylabel={Recall after one week (\%)},
  nodes near coords,every node near coord/.append style={font=\fontsize{14}{16}\selectfont},
  axis lines*=left,ymajorgrids,grid style={color=lsRule},enlarge x limits=0.5,
  tick label style={font=\fontsize{14}{16}\selectfont},label style={font=\fontsize{14}{16}\selectfont}]
\addplot[fill=lsAccent,draw=none] coordinates {(Text only,10) (Text + figure,65)};
\end{axis}\end{tikzpicture}\par
{\fontsize{\lsSizeCaption}{\lsLeadCaption}\selectfont\color{lsMuted}Illustrative numbers; picture-superiority effect (Medina, 2008).}}
\at{560}{\lsContentTop}{344}{\point{Rule}{At least one figure, diagram, chart, table or video per content slide.}\vspace{12pt}
\point{Budget}{About 50 English words or 80 CJK characters; split instead of shrinking.}}
\takeaway{check.py fails any content slide without a visual.}
\end{frame}
