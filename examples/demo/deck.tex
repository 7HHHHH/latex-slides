  \draw[arr] (yaml.east) -| (pptx.north);

  \draw[arr] (pptx.south) |- (chk.east);
\end{tikzpicture}}
\begin{lsgrid}[344]{4}
\card{\stat{6.0 s}{full build of these 8 pages}}{CPU only, no GPU.}
\card{\stat{18}{automatic checks}}{Glyphs to pixel diffs.}
\card{\stat{1}{place to edit}}{Notes live in deck.yaml.}
\card{\stat{0}{files overwritten}}{Rebuilds add \_v2, \_v3.}
\end{lsgrid}
\takeaway{Edit the LaTeX, rebuild, and the PDF and PPTX cannot drift apart.}
\end{frame}

\lsChapter{03}
\begin{frame}{Checks run before anyone looks at a slide}\slideid{checks}
\lead{check.py fails the build on hard problems and warns on soft ones.}
\at{56}{148}{460}{\renewcommand{\arraystretch}{1.22}\fontsize{14}{17}\selectfont
\begin{tabular}{@{}p{70bp}p{240bp}l@{}}
\toprule
\textbf{Area} & \textbf{What is checked} & \textbf{Level}\\
\midrule
LaTeX & glyphs, overfull boxes, refs & \textcolor{lsAccent}{FAIL}\\
Layout & safe area, overlaps & \textcolor{lsAccent}{FAIL}\\
Visuals & one per slide, text budget & \textcolor{lsAccent}{FAIL}\\
Density & near-empty slides & WARN\\
Package & SVG, notes, links, video & \textcolor{lsAccent}{FAIL}\\
Render & PPTX vs.\ PDF pixels & \textcolor{lsAccent}{FAIL}\\
Timing & seconds vs.\ talk length & \textcolor{lsAccent}{FAIL}\\
\bottomrule
\end{tabular}}
